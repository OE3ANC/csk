cmake_minimum_required(VERSION 3.20.0)

macro(csk_zephyr_ztest_relative)
    set(__SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR})
    while(__SOURCE_DIR)
        get_filename_component(__SOURCE_DIR ${__SOURCE_DIR} DIRECTORY)
        get_filename_component(LAST_DIR ${__SOURCE_DIR} NAME)
        if ("tests" STREQUAL LAST_DIR)
            get_filename_component(RELATIVE_PATH ${__SOURCE_DIR} DIRECTORY)
            get_filename_component(ABSOLUTE_PATH2 ${CMAKE_CURRENT_SOURCE_DIR} ABSOLUTE)
            file(RELATIVE_PATH RELATIVE_PATH2 ${RELATIVE_PATH} ${ABSOLUTE_PATH2})
            set(CSK_CURRENT_ZTEST_RELATIVE_PATH ${RELATIVE_PATH2})
            break()
        endif()
    endwhile()
endmacro()

macro(csk_zephyr_ztest_amend)
    set(ZEPHYR_TESTS_DIR $ENV{ZEPHYR_BASE}/${CSK_CURRENT_ZTEST_RELATIVE_PATH})
    get_filename_component(CUSTOM_TEST_PROJECT_NAME ${ZEPHYR_TESTS_DIR} NAME)
    set(CONF_FILE ${ZEPHYR_TESTS_DIR}/prj.conf)
    find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
    project(${CUSTOM_TEST_PROJECT_NAME})
    add_subdirectory(${ZEPHYR_TESTS_DIR} ${CUSTOM_TEST_PROJECT_NAME})
endmacro()

csk_zephyr_ztest_relative()
csk_zephyr_ztest_amend()
